     5→     5→// Page publique mobile qu'un client voit après avoir scanné le
     6→     6→// QR code d'un commerce. Sans authentification, 3 étapes :
     7→     7→//   step1 : la note (étoiles) → step2 : tags dynamiques → step3 : commentaire + envoi
     8→     8→// Machine à états : loading → error | step1 → step2 → step3 → success

    24→    24→import { buildWaLink } from "@/lib/whatsapp"
    25→    25→import { cn } from "@/lib/utils"
    26→    26→import { Button } from "@/components/ui/button"

    45→    45→  ville?: string | null
    46→    46→  logoUrl?: string | null
    47→    47→  googleUrl?: string | null

    63→    63→// En-tête commun : logo circulaire + nom du commerce + catégorie · ville
    64→    64→function EnteteCommerce({ pro, avecMarge }: { pro: ScanData; avecMarge: boolean }) {
    65→    65→  // Fallback local si le logo distant ne charge pas
    66→    66→  const [logoErreur, setLogoErreur] = useState(false)
    67→    67→  const initiale = pro.nomCommerce.charAt(0).toUpperCase()
    68→    68→
    69→    69→  return (
    70→    70→    <div className={cn("mb-6 flex items-center gap-3", avecMarge && "pr-12")}>
    71→    71→      {pro.logoUrl && !logoErreur ? (
    72→    72→        <img
    73→    73→          src={pro.logoUrl}
    74→    74→          alt={`Logo ${pro.nomCommerce}`}
    75→    75→          className="size-14 shrink-0 rounded-full border border-stone-200 object-cover"
    76→    76→          onError={() => setLogoErreur(true)}
    77→    77→        />
    78→    78→      ) : (
    79→    79→        <div
    80→    80→          aria-hidden
    81→    81→          className="flex size-14 shrink-0 items-center justify-center rounded-full border border-primary/20 bg-primary/10 text-xl font-bold text-primary"
    82→    82→        >
    83→    83→          {initiale}
    84→    84→        </div>
    85→    85→      )}
    86→    86→      <div className="min-w-0">
    87→    87→        <p className="truncate text-lg font-bold leading-tight text-stone-900">{pro.nomCommerce}</p>
    88→    88→        {(pro.categorie || pro.ville) && (
    89→    89→          <p className="truncate text-sm text-muted-foreground">
    90→    90→            {[pro.categorie, pro.ville].filter(Boolean).join(" · ")}
    91→    91→          </p>
    92→    92→        )}
    93→    93→      </div>
    94→    94→    </div>
    95→    95→  )
    96→    96→}

    98→    98→// ---------- Composant principal ----------
    99→    99→
   100→   100→export function ScanView({ token, onExit }: { token: string; onExit?: () => void }) {
   101→   101→  const { toast } = useToast()
   102→   102→
   103→   103→  // Machine à états
   104→   104→  const [phase, setPhase] = useState<Phase>("loading")
   105→   105→  const [pro, setPro] = useState<ScanData | null>(null)
   106→   106→
   107→   107→  // Réponses du client
   108→   108→  const [note, setNote] = useState(0)
   109→   109→  const [tags, setTags] = useState<string[]>([])
   110→   110→  const [employeId, setEmployeId] = useState("") // "" = non précisé, "autre" = autre / je ne sais pas
   111→   111→  const [personnelLibre, setPersonnelLibre] = useState("") // input texte si aucun employé enregistré
   112→   112→  const [commentaire, setCommentaire] = useState("")
   113→   113→
   114→   114→  // Animation d'étoile + envoi
   115→   115→  const [pulse, setPulse] = useState<number | null>(null)
   116→   116→  const [soumission, setSoumission] = useState<CanalEnvoi | null>(null)
   117→   117→  const [dernierCanal, setDernierCanal] = useState<CanalEnvoi>("whatsapp")
   118→   118→  const [erreurEnvoi, setErreurEnvoi] = useState<string | null>(null)
   119→   119→
   120→   120→  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   121→   121→  useEffect(() => {
   122→   122→    // Nettoyage du timer de transition étoile → étape 2
   123→   123→    return () => {
   124→   124→      if (timerRef.current) clearTimeout(timerRef.current)
   125→   125→    }
   126→   126→  }, [])
   127→   127→
   128→   128→  const estPositif = note >= 4
   129→   129→
   130→   130→  // ---------- Chargement des infos du commerce ----------
   131→   131→  const chargerScan = useCallback(async () => {
   132→   132→    setPhase("loading")
   133→   133→    setErreurEnvoi(null)
   134→   134→    try {
   135→   135→      const res = await fetch(`/api/public/scan?token=${encodeURIComponent(token)}`, {
   136→   136→        cache: "no-store",
   137→   137→      })
   138→   138→      const json = await res.json().catch(() => null)
   139→   139→      if (!res.ok || !json?.ok) throw new Error(json?.error || "QR code invalide ou expiré")
   140→   140→      setPro(json.data as ScanData)
   141→   141→      setPhase("step1")
   142→   142→    } catch {
   143→   143→      setPro(null)
   144→   144→      setPhase("error")
   145→   145→    }
   146→   146→  }, [token])
   147→   147→
   148→   148→  useEffect(() => {
   149→   149→    chargerScan()
   150→   150→  }, [chargerScan])
   151→   151→
   152→   152→  // ---------- Étape 1 : la note ----------
   153→   153→  function choisirNote(n: number) {
   154→   154→    setNote(n)
   155→   155→    // Réinitialise les choix dépendants de la note (tags positifs ↔ négatifs)
   156→   156→    setTags([])
   157→   157→    setEmployeId("")
   158→   158→    setPersonnelLibre("")
   159→   159→    setPulse(n)
   160→   160→    if (timerRef.current) clearTimeout(timerRef.current)
   161→   161→    // Petite animation pulse puis passage auto à l'étape 2
   162→   162→    timerRef.current = setTimeout(() => setPhase("step2"), 350)
   163→   163→  }
   164→   164→
   165→   165→  // Navigation clavier dans le radiogroup des étoiles
   166→   166→  function etoileClavier(e: React.KeyboardEvent, valeurCourante: number) {
   167→   167→    if (e.key === "ArrowRight" || e.key === "ArrowUp") {
   168→   168→      e.preventDefault()
   169→   169→      const n = Math.min(5, valeurCourante + 1)
   170→   170→      if (n >= 1) choisirNote(n)
   171→   171→    } else if (e.key === "ArrowLeft" || e.key === "ArrowDown") {
   172→   172→      e.preventDefault()
   173→   173→      choisirNote(Math.max(1, (valeurCourante || 2) - 1))
   174→   174→    }
   175→   175→  }
   176→   176→
   177→   177→  // ---------- Étape 2 : les tags ----------
   178→   178→  const tagsDisponibles = estPositif ? POSITIVE_TAGS : NEGATIVE_TAGS
   179→   179→
   180→   180→  function basculerTag(id: string) {
   181→   181→    setTags((prev) => (prev.includes(id) ? prev.filter((t) => t !== id) : [...prev, id]))
   182→   182→  }
   183→   183→
   184→   184→  // ---------- Soumission ----------
   185→   185→  async function envoyer(canal: CanalEnvoi) {
   186→   186→    if (!pro || soumission) return
   187→   187→    setDernierCanal(canal)
   188→   188→    setSoumission(canal)
   189→   189→    setErreurEnvoi(null)
   190→   190→    try {
   191→   191→      // Sécurité : on ne transmet que des ids de tags connus
   192→   192→      const tagsValides = tags.filter((id) => ALL_TAGS.some((t) => t.id === id))
   193→   193→      const res = await fetch("/api/public/retour", {
   194→   194→        method: "POST",
   195→   195→        headers: { "Content-Type": "application/json" },
   196→   196→        body: JSON.stringify({
   197→   197→          token,
   198→   198→          note,
   199→   199→          tags: tagsValides,
   200→   200→          commentaire: commentaire.trim() || null,
   201→   201→          // Selon la configuration du commerce : employeId OU personnel libre
   202→   202→          employeId: pro.employes.length > 0 && employeId && employeId !== "autre" ? employeId : null,
   203→   203→          personnelMentionne: pro.employes.length === 0 ? personnelLibre.trim() || null : null,
   204→   204→          canalChoisi: canal,
   205→   205→        }),
   206→   206→      })
   207→   207→      const json = await res.json().catch(() => null)
   208→   208→      if (!res.ok || !json?.ok) throw new Error(json?.error || "Erreur lors de l'envoi du retour")
   209→   209→
   210→   210→      // Ouvre WhatsApp avec le message pré-rempli ; si popup bloqué → navigation directe
   211→   211→      if (canal === "whatsapp" && pro.whatsapp) {
   212→   212→        const employeNom = pro.employes.find((e) => e.id === employeId)?.nom ?? null
   213→   213→        const lien = buildWaLink(
   214→   214→          {
   215→   215→            nomCommerce: pro.nomCommerce,
   216→   216→            whatsapp: pro.whatsapp,
   217→   217→            messageWhatsapp: pro.messageWhatsapp,
   218→   218→          },
   219→   219→          {
   220→   220→            note,
   221→   221→            tags: tagsValides,
   222→   222→            commentaire: commentaire.trim() || null,
   223→   223→            personnelMentionne: pro.employes.length === 0 ? personnelLibre.trim() || null : null,
   224→   224→          },
   225→   225→          employeNom
   226→   226→        )
   227→   227→        const fenetre = window.open(lien, "_blank")
   228→   228→        if (!fenetre) window.location.href = lien
   229→   229→      }
   230→   230→
   231→   231→      setPhase("success")
   232→   232→    } catch (e) {
   233→   233→      const message = e instanceof Error ? e.message : "Erreur lors de l'envoi du retour"
   234→   234→      setErreurEnvoi(message)
   235→   235→      toast({ title: "Envoi impossible", description: message, variant: "destructive" })
   236→   236→    } finally {
   237→   237→      setSoumission(null)
   238→   238→    }
   239→   239→  }
   240→   240→
   241→   241→  // ---------- Écrans plein écran : chargement ----------
   242→   242→  if (phase === "loading") {
   243→   243→    return (
   244→   244→      <div className="flex min-h-dvh flex-col items-center justify-center gap-3 bg-stone-50 p-6">
   245→   245→        <Loader2 className="size-10 animate-spin text-primary" aria-hidden />
   246→   246→        <p className="text-sm text-muted-foreground">Chargement du formulaire de retour…</p>
   247→   247→      </div>
   248→   248→    )
   249→   249→  }
   250→   250→
   251→   251→  // Indice d'étape pour la barre de progression (3 segments)
   252→   252→  const etapeIdx = phase === "step1" ? 0 : phase === "step2" ? 1 : 2
   253→   253→
   254→   254→  return (
   255→   255→    <div className="flex min-h-dvh items-center justify-center bg-stone-50 p-3 sm:p-6">
   256→   256→      <div className="w-full max-w-md">
   257→   257→          <div className="relative overflow-hidden rounded-3xl border border-stone-200 bg-white shadow-lg">
   258→   258→            {/* Barre de progression fine : 3 segments */}
   259→   259→            {phase !== "error" && (
   260→   260→              <div className="flex gap-1.5 px-6 pt-5 sm:px-8" aria-hidden>
   261→   261→                {[0, 1, 2].map((s) => (
   262→   262→                  <div
   263→   263→                    key={s}
   264→   264→                    className={cn(
   265→   265→                      "h-1.5 flex-1 rounded-full transition-colors duration-300",
   266→   266→                      phase === "success" || s <= etapeIdx ? "bg-primary" : "bg-stone-200"
   267→   267→                    )}
   268→   268→                  />
   269→   269→                ))}
   270→   270→              </div>
   271→   271→            )}
   272→   272→
   273→   273→            {/* Bouton discret Quitter (étapes 1 à 3) */}
   274→   274→            {onExit && (phase === "step1" || phase === "step2" || phase === "step3") && (
   275→   275→              <Button
   276→   276→                variant="ghost"
   277→   277→                size="icon"
   278→   278→                onClick={onExit}
   279→   279→                aria-label="Quitter et revenir en arrière"
   280→   280→                className="absolute right-3 top-4 z-10 size-11 rounded-full text-stone-400 hover:text-stone-600"
   281→   281→              >
   282→   282→                <X className="size-5" aria-hidden />
   283→   283→              </Button>
   284→   284→            )}
   285→   285→
   286→   286→            <div className="p-6 sm:p-8">
   287→   287→              {/* En-tête du commerce sur toutes les étapes du formulaire */}
   288→   288→              {pro && phase !== "error" && <EnteteCommerce pro={pro} avecMarge={phase !== "success"} />}
   289→   289→
   290→   290→              <AnimatePresence mode="wait" initial={false}>
   291→   291→                <motion.div
   292→   292→                  key={phase}
   293→   293→                  variants={variantesEtape}
   294→   294→                  initial="entree"
   295→   295→                  animate="visible"
   296→   296→                  exit="sortie"
   297→   297→                  transition={{ duration: 0.25, ease: "easeOut" }}
   298→   298→                >
   299→   299→                  {/* ---------- ERREUR : token invalide ---------- */}
   300→   300→                  {phase === "error" && (
   301→   301→                    <div className="flex flex-col items-center gap-4 py-8 text-center">
   302→   302→                      <div className="flex size-16 items-center justify-center rounded-full bg-rose-50">
   303→   303→                        <AlertCircle className="size-9 text-rose-500" aria-hidden />
   304→   304→                      </div>
   305→   305→                      <div>
   306→   306→                        <h1 className="text-xl font-bold text-stone-900">QR code invalide ou expiré</h1>
   307→   307→                        <p className="mt-1 text-sm text-muted-foreground">
   308→   308→                          Ce lien de scan n&apos;est plus actif. Vérifiez le QR code affiché chez le
   309→   309→                          commerçant.
   310→   310→                        </p>
   311→   311→                      </div>
   312→   312→                      <Button
   313→   313→                        onClick={chargerScan}
   314→   314→                        className="h-12 min-w-40 text-base font-semibold"
   315→   315→                        aria-label="Réessayer le chargement"
   316→   316→                      >
   317→   317→                        Réessayer
   318→   318→                      </Button>
   319→   319→                      {onExit && (
   320→   320→                        <Button
   321→   321→                          variant="ghost"
   322→   322→                          onClick={onExit}
   323→   323→                          className="min-h-11 text-sm text-muted-foreground"
   324→   324→                        >
   325→   325→                          Retour à l&apos;accueil
   326→   326→                        </Button>
   327→   327→                      )}
   328→   328→                    </div>
   329→   329→                  )}
   330→   330→
   331→   331→                  {/* ---------- ÉTAPE 1 : la note ---------- */}
   332→   332→                  {phase === "step1" && pro && (
   333→   333→                    <div className="flex flex-col items-center text-center">
   334→   334→                      <h1 className="text-balance text-2xl font-bold leading-snug text-stone-900">
   335→   335→                        Comment s&apos;est passée votre visite chez{" "}
   336→   336→                        <span className="text-primary">{pro.nomCommerce}</span> ?
   337→   337→                      </h1>
   338→   338→
   339→   339→                      <div
   340→   340→                        role="radiogroup"
   341→   341→                        aria-label="Votre note de 1 à 5 étoiles"
   342→   342→                        className="my-8 flex justify-center gap-1 sm:gap-2"
   343→   343→                      >
   344→   344→                        {[1, 2, 3, 4, 5].map((i) => (
   345→   345→                          <motion.button
   346→   346→                            key={i}
   347→   347→                            type="button"
   348→   348→                            role="radio"
   349→   349→                            aria-checked={note === i}
   350→   350→                            aria-label={`Noter ${i} sur 5`}
   351→   351→                            onClick={() => choisirNote(i)}
   352→   352→                            onKeyDown={(e) => etoileClavier(e, note)}
   353→   353→                            whileHover={{ scale: 1.12 }}
   354→   354→                            whileTap={{ scale: 0.88 }}
   355→   355→                            animate={pulse === i ? { scale: [1, 1.3, 1] } : { scale: 1 }}
   356→   356→                            transition={{ duration: 0.35, ease: "easeInOut" }}
   357→   357→                            className="flex size-14 items-center justify-center rounded-full outline-none focus-visible:ring-2 focus-visible:ring-ring focus-visible:ring-offset-2"
   358→   358→                          >
   359→   359→                            <Star
   360→   360→                              aria-hidden
   361→   361→                              strokeWidth={1.5}
   362→   362→                              className={cn(
   363→   363→                                "size-12 transition-colors duration-150",
   364→   364→                                i <= note ? "fill-amber-400 text-amber-400" : "fill-stone-200 text-stone-300"
   365→   365→                              )}
   366→   366→                            />
   367→   367→                          </motion.button>
   368→   368→                        ))}
   369→   369→                      </div>
   370→   370→
   371→   371→                      <p className="text-sm text-muted-foreground">30 secondes suffisent, merci !</p>
   372→   372→                    </div>
   373→   373→                  )}
   374→   374→
   375→   375→                  {/* ---------- ÉTAPE 2 : les tags dynamiques ---------- */}
   376→   376→                  {phase === "step2" && pro && (
   377→   377→                    <div>
   378→   378→                      <h1 className="text-2xl font-bold text-stone-900">
   379→   379→                        {estPositif ? "Qu'est-ce qui vous a plu ?" : "Qu'est-ce qui s'est passé ?"}
   380→   380→                      </h1>
   381→   381→                      <p className="mt-1 text-sm text-muted-foreground">
   382→   382→                        Sélectionnez tout ce qui s&apos;applique (optionnel)
   383→   383→                      </p>
   384→   384→
   385→   385→                      <div
   386→   386→                        role="group"
   387→   387→                        aria-label={estPositif ? "Points positifs" : "Points à améliorer"}
   388→   388→                        className="mt-5 grid grid-cols-2 gap-3"
   389→   389→                      >
   390→   390→                        {tagsDisponibles.map((t) => {
   391→   391→                          const actif = tags.includes(t.id)
   392→   392→                          return (
   393→   393→                            <button
   394→   394→                              key={t.id}
   395→   395→                              type="button"
   396→   396→                              aria-pressed={actif}
   397→   397→                              onClick={() => basculerTag(t.id)}
   398→   398→                              className={cn(
   399→   399→                                "flex min-h-12 min-w-0 items-center gap-2 rounded-2xl border-2 px-3 py-3 text-left text-sm font-medium transition-all active:scale-95 sm:text-base",
   400→   400→                                "outline-none focus-visible:ring-2 focus-visible:ring-ring focus-visible:ring-offset-2",
   401→   401→                                actif
   402→   402→                                  ? estPositif
   403→   403→                                    ? "border-primary bg-primary/10 text-primary"
   404→   404→                                    : "border-rose-500 bg-rose-50 text-rose-700"
   405→   405→                                  : "border-stone-200 bg-white text-stone-700 hover:border-stone-300 hover:bg-stone-50"
   406→   406→                              )}
   407→   407→                            >
   408→   408→                              <span aria-hidden className="text-lg leading-none">
   409→   409→                                {t.emoji}
   410→   410→                              </span>
   411→   411→                              <span className="min-w-0 whitespace-nowrap leading-tight">{t.label}</span>
   412→   412→                            </button>
   413→   413→                          )
   414→   414→                        })}
   415→   415→                      </div>
   416→   416→
   417→   417→                      {/* Champ optionnel : personnel concerné */}
   418→   418→                      <div className="mt-5">
   419→   419→                        <label
   420→   420→                          htmlFor="personnel"
   421→   421→                          className="mb-1.5 block text-sm font-medium text-stone-700"
   422→   422→                        >
   423→   423→                          Personnel concerné{" "}
   424→   424→                          <span className="font-normal text-muted-foreground">(optionnel)</span>
   425→   425→                        </label>
   426→   426→                        {pro.employes.length > 0 ? (
   427→   427→                          <Select value={employeId} onValueChange={setEmployeId}>
   428→   428→                            <SelectTrigger
   429→   429→                              id="personnel"
   430→   430→                              className="h-12 w-full text-base"
   431→   431→                              aria-label="Sélectionner le personnel concerné"
   432→   432→                            >
   433→   433→                              <SelectValue placeholder="Personnel concerné (optionnel)" />
   434→   434→                            </SelectTrigger>
   435→   435→                            <SelectContent>
   436→   436→                              {pro.employes.map((e) => (
