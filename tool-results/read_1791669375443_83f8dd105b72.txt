   209→   209→// Compteur animé déclenché quand la stat entre dans le viewport
   210→   210→function Compteur({ fin, decimales = 0 }: { fin: number; decimales?: number }) {
   211→   211→  const ref = useRef<HTMLSpanElement>(null)
   212→   212→  const visible = useInView(ref, { once: true, amount: 0.6 })
   213→   213→  const [valeur, setValeur] = useState(0)
   214→   214→
   215→   215→  useEffect(() => {
   216→   216→    if (!visible) return
   217→   217→    const duree = 1400
   218→   218→    const debut = performance.now()
   219→   219→    let id = 0
   220→   220→    const boucle = (t: number) => {
   221→   221→      const p = Math.min((t - debut) / duree, 1)
   222→   222→      setValeur(fin * (1 - Math.pow(1 - p, 3))) // easing easeOutCubic
   223→   223→      if (p < 1) id = requestAnimationFrame(boucle)
   224→   224→    }
   225→   225→    id = requestAnimationFrame(boucle)
   226→   226→    return () => cancelAnimationFrame(id)
   227→   227→  }, [visible, fin])
   228→   228→
   229→   229→  return (
   230→   230→    <span ref={ref}>
   231→   231→      {Intl.NumberFormat("fr-FR", {
   232→   232→        minimumFractionDigits: decimales,
   233→   233→        maximumFractionDigits: decimales,
   234→   234→      }).format(valeur)}
   235→   235→    </span>
   236→   236→  )
   237→   237→}

   239→   239→// Génère une grille 21x21 pseudo-aléatoire DÉTERMINISTE (même rendu serveur/client)
   240→   240→function genererGrilleQr(taille: number, graine: number): boolean[][] {
   241→   241→  let etat = graine
   242→   242→  const alea = () => {
   243→   243→    etat = (etat * 1103515245 + 12345) % 2147483648
   244→   244→    return etat / 2147483648
   245→   245→  }
   246→   246→  const grille: boolean[][] = Array.from({ length: taille }, () =>
   247→   247→    Array<boolean>(taille).fill(false)
   248→   248→  )
   249→   249→  // Les 3 motifs de repérage (coins) sont dessinés à part
   250→   250→  const estFinder = (r: number, c: number) =>
   251→   251→    (r < 7 && c < 7) || (r < 7 && c >= taille - 7) || (r >= taille - 7 && c < 7)
   252→   252→
   253→   253→  for (let r = 0; r < taille; r++) {
   254→   254→    for (let c = 0; c < taille; c++) {
   255→   255→      if (estFinder(r, c)) continue
   256→   256→      if (r === 6 || c === 6) {
   257→   257→        grille[r][c] = (r + c) % 2 === 0 // motif de synchronisation
   258→   258→      } else {
   259→   259→        grille[r][c] = alea() > 0.52
   260→   260→      }
   261→   261→    }
   262→   262→  }
   263→   263→  return grille
   264→   264→}

   279→   279→      <g fill="currentColor">
   280→   280→        {grille.map((ligne, r) =>
   281→   281→          ligne.map((plein, c) =>
   282→   282→            plein ? <rect key={`${r}-${c}`} x={c} y={r} width={1} height={1} /> : null
   283→   283→          )
   284→   284→        )}
   285→   285→      </g>
   286→   286→      {/* Motifs de repérage : carré plein, liseré blanc, carré central */}
   287→   287→      {[
   288→   288→        [0, 0],
   289→   289→        [0, taille - 7],
   290→   290→        [taille - 7, 0],
   291→   291→      ].map(([r, c]) => (
   292→   292→        <g key={`finder-${r}-${c}`}>
   293→   293→          <rect x={c} y={r} width={7} height={7} rx={1.5} fill="currentColor" />
   294→   294→          <rect x={c + 1} y={r + 1} width={5} height={5} rx={1} fill="#ffffff" />
   295→   295→          <rect x={c + 2} y={r + 2} width={3} height={3} rx={0.5} fill="currentColor" />
   296→   296→        </g>
   297→   297→      ))}
   298→   298→    </svg>
