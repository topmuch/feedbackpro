     1→"use client"
     2→
     3→// ============================================================
     4→// FeedbackPro - ScanView
     5→// Page publique mobile qu'un client voit après avoir scanné le
     6→// QR code d'un commerce. Sans authentification, 3 étapes :
     7→//   step1 : la note (étoiles) → step2 : tags dynamiques → step3 : commentaire + envoi
     8→// Machine à états : loading → error | step1 → step2 → step3 → success
     9→// ============================================================
    10→
    11→import { useCallback, useEffect, useRef, useState } from "react"
    12→import { AnimatePresence, motion } from "framer-motion"
    13→import {
    14→  AlertCircle,
    15→  ArrowLeft,
    16→  Check,
    17→  Loader2,
    18→  MessageCircle,
    19→  ShieldCheck,
    20→  Star,
    21→  X,
    22→} from "lucide-react"
    23→import { POSITIVE_TAGS, NEGATIVE_TAGS, ALL_TAGS, tagLabel } from "@/lib/constants"
    24→import { buildWaLink } from "@/lib/whatsapp"
    25→import { cn } from "@/lib/utils"
    26→import { Button } from "@/components/ui/button"
    27→import { Input } from "@/components/ui/input"
    28→import { Textarea } from "@/components/ui/textarea"
    29→import {
    30→  Select,
    31→  SelectContent,
    32→  SelectItem,
    33→  SelectTrigger,
    34→  SelectValue,
    35→} from "@/components/ui/select"
    36→import { useToast } from "@/hooks/use-toast"
    37→
    38→// ---------- Types locaux ----------
    39→
    40→type EmployeScan = { id: string; nom: string; poste?: string | null }
    41→
    42→type ScanData = {
    43→  nomCommerce: string
    44→  categorie?: string | null
    45→  ville?: string | null
    46→  logoUrl?: string | null
    47→  googleUrl?: string | null
    48→  whatsapp?: string | null
    49→  messageWhatsapp?: string | null
    50→  employes: EmployeScan[]
    51→}
    52→
    53→type Phase = "loading" | "error" | "step1" | "step2" | "step3" | "success"
    54→type CanalEnvoi = "whatsapp" | "aucun"
    55→
    56→// Transition fade + slide entre les étapes
    57→const variantesEtape = {
    58→  entree: { opacity: 0, x: 32 },
    59→  visible: { opacity: 1, x: 0 },
    60→  sortie: { opacity: 0, x: -32 },
    61→}
    62→
    63→// En-tête commun : logo circulaire + nom du commerce + catégorie · ville
    64→function EnteteCommerce({ pro, avecMarge }: { pro: ScanData; avecMarge: boolean }) {
    65→  // Fallback local si le logo distant ne charge pas
    66→  const [logoErreur, setLogoErreur] = useState(false)
    67→  const initiale = pro.nomCommerce.charAt(0).toUpperCase()
    68→
    69→  return (
    70→    <div className={cn("mb-6 flex items-center gap-3", avecMarge && "pr-12")}>
    71→      {pro.logoUrl && !logoErreur ? (
    72→        <img
    73→          src={pro.logoUrl}
    74→          alt={`Logo ${pro.nomCommerce}`}
    75→          className="size-14 shrink-0 rounded-full border border-stone-200 object-cover"
    76→          onError={() => setLogoErreur(true)}
    77→        />
    78→      ) : (
    79→        <div
    80→          aria-hidden
    81→          className="flex size-14 shrink-0 items-center justify-center rounded-full border border-primary/20 bg-primary/10 text-xl font-bold text-primary"
    82→        >
    83→          {initiale}
    84→        </div>
    85→      )}
    86→      <div className="min-w-0">
    87→        <p className="truncate text-lg font-bold leading-tight text-stone-900">{pro.nomCommerce}</p>
    88→        {(pro.categorie || pro.ville) && (
    89→          <p className="truncate text-sm text-muted-foreground">
    90→            {[pro.categorie, pro.ville].filter(Boolean).join(" · ")}
    91→          </p>
    92→        )}
    93→      </div>
    94→    </div>
    95→  )
    96→}
    97→
    98→// ---------- Composant principal ----------
    99→
   100→export function ScanView({ token, onExit }: { token: string; onExit?: () => void }) {
   101→  const { toast } = useToast()
   102→
   103→  // Machine à états
   104→  const [phase, setPhase] = useState<Phase>("loading")
   105→  const [pro, setPro] = useState<ScanData | null>(null)
   106→
   107→  // Réponses du client
   108→  const [note, setNote] = useState(0)
   109→  const [tags, setTags] = useState<string[]>([])
   110→  const [employeId, setEmployeId] = useState("") // "" = non précisé, "autre" = autre / je ne sais pas
   111→  const [personnelLibre, setPersonnelLibre] = useState("") // input texte si aucun employé enregistré
   112→  const [commentaire, setCommentaire] = useState("")
   113→
   114→  // Animation d'étoile + envoi
   115→  const [pulse, setPulse] = useState<number | null>(null)
   116→  const [soumission, setSoumission] = useState<CanalEnvoi | null>(null)
   117→  const [dernierCanal, setDernierCanal] = useState<CanalEnvoi>("whatsapp")
   118→  const [erreurEnvoi, setErreurEnvoi] = useState<string | null>(null)
   119→
   120→  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   121→  useEffect(() => {
   122→    // Nettoyage du timer de transition étoile → étape 2
   123→    return () => {
   124→      if (timerRef.current) clearTimeout(timerRef.current)
   125→    }
   126→  }, [])
   127→
   128→  const estPositif = note >= 4
   129→
   130→  // ---------- Chargement des infos du commerce ----------
   131→  const chargerScan = useCallback(async () => {
   132→    setPhase("loading")
   133→    setErreurEnvoi(null)
   134→    try {
   135→      const res = await fetch(`/api/public/scan?token=${encodeURIComponent(token)}`, {
   136→        cache: "no-store",
   137→      })
   138→      const json = await res.json().catch(() => null)
   139→      if (!res.ok || !json?.ok) throw new Error(json?.error || "QR code invalide ou expiré")
   140→      setPro(json.data as ScanData)
   141→      setPhase("step1")
   142→    } catch {
   143→      setPro(null)
   144→      setPhase("error")
   145→    }
   146→  }, [token])
   147→
   148→  useEffect(() => {
   149→    chargerScan()
   150→  }, [chargerScan])
   151→
   152→  // ---------- Étape 1 : la note ----------
   153→  function choisirNote(n: number) {
   154→    setNote(n)
   155→    // Réinitialise les choix dépendants de la note (tags positifs ↔ négatifs)
   156→    setTags([])
   157→    setEmployeId("")
   158→    setPersonnelLibre("")
   159→    setPulse(n)
   160→    if (timerRef.current) clearTimeout(timerRef.current)
   161→    // Petite animation pulse puis passage auto à l'étape 2
   162→    timerRef.current = setTimeout(() => setPhase("step2"), 350)
   163→  }
   164→
   165→  // Navigation clavier dans le radiogroup des étoiles
   166→  function etoileClavier(e: React.KeyboardEvent, valeurCourante: number) {
   167→    if (e.key === "ArrowRight" || e.key === "ArrowUp") {
   168→      e.preventDefault()
   169→      const n = Math.min(5, valeurCourante + 1)
   170→      if (n >= 1) choisirNote(n)
   171→    } else if (e.key === "ArrowLeft" || e.key === "ArrowDown") {
   172→      e.preventDefault()
   173→      choisirNote(Math.max(1, (valeurCourante || 2) - 1))
   174→    }
   175→  }
   176→
   177→  // ---------- Étape 2 : les tags ----------
   178→  const tagsDisponibles = estPositif ? POSITIVE_TAGS : NEGATIVE_TAGS
   179→
   180→  function basculerTag(id: string) {
   181→    setTags((prev) => (prev.includes(id) ? prev.filter((t) => t !== id) : [...prev, id]))
   182→  }
   183→
   184→  // ---------- Soumission ----------
   185→  async function envoyer(canal: CanalEnvoi) {
   186→    if (!pro || soumission) return
   187→    setDernierCanal(canal)
   188→    setSoumission(canal)
   189→    setErreurEnvoi(null)
   190→    try {
   191→      // Sécurité : on ne transmet que des ids de tags connus
   192→      const tagsValides = tags.filter((id) => ALL_TAGS.some((t) => t.id === id))
   193→      const res = await fetch("/api/public/retour", {
   194→        method: "POST",
   195→        headers: { "Content-Type": "application/json" },
   196→        body: JSON.stringify({
   197→          token,
   198→          note,
   199→          tags: tagsValides,
   200→          commentaire: commentaire.trim() || null,
   201→          // Selon la configuration du commerce : employeId OU personnel libre
   202→          employeId: pro.employes.length > 0 && employeId && employeId !== "autre" ? employeId : null,
   203→          personnelMentionne: pro.employes.length === 0 ? personnelLibre.trim() || null : null,
   204→          canalChoisi: canal,
   205→        }),
   206→      })
   207→      const json = await res.json().catch(() => null)
   208→      if (!res.ok || !json?.ok) throw new Error(json?.error || "Erreur lors de l'envoi du retour")
   209→
   210→      // Ouvre WhatsApp avec le message pré-rempli ; si popup bloqué → navigation directe
   211→      if (canal === "whatsapp" && pro.whatsapp) {
   212→        const employeNom = pro.employes.find((e) => e.id === employeId)?.nom ?? null
   213→        const lien = buildWaLink(
   214→          {
   215→            nomCommerce: pro.nomCommerce,
   216→            whatsapp: pro.whatsapp,
   217→            messageWhatsapp: pro.messageWhatsapp,
   218→          },
   219→          {
   220→            note,
   221→            tags: tagsValides,
   222→            commentaire: commentaire.trim() || null,
   223→            personnelMentionne: pro.employes.length === 0 ? personnelLibre.trim() || null : null,
   224→          },
   225→          employeNom
   226→        )
   227→        const fenetre = window.open(lien, "_blank")
   228→        if (!fenetre) window.location.href = lien
   229→      }
   230→
   231→      setPhase("success")
   232→    } catch (e) {
   233→      const message = e instanceof Error ? e.message : "Erreur lors de l'envoi du retour"
   234→      setErreurEnvoi(message)
   235→      toast({ title: "Envoi impossible", description: message, variant: "destructive" })
   236→    } finally {
   237→      setSoumission(null)
   238→    }
   239→  }
   240→
   241→  // ---------- Écrans plein écran : chargement ----------
   242→  if (phase === "loading") {
   243→    return (
   244→      <div className="flex min-h-dvh flex-col items-center justify-center gap-3 bg-stone-50 p-6">
   245→        <Loader2 className="size-10 animate-spin text-primary" aria-hidden />
   246→        <p className="text-sm text-muted-foreground">Chargement du formulaire de retour…</p>
   247→      </div>
   248→    )
   249→  }
   250→
   251→  // Indice d'étape pour la barre de progression (3 segments)
   252→  const etapeIdx = phase === "step1" ? 0 : phase === "step2" ? 1 : 2
   253→
   254→  return (
   255→    <div className="flex min-h-dvh items-center justify-center bg-stone-50 p-3 sm:p-6">
   256→      <div className="w-full max-w-md">
   257→          <div className="relative overflow-hidden rounded-3xl border border-stone-200 bg-white shadow-lg">
   258→            {/* Barre de progression fine : 3 segments */}
   259→            {phase !== "error" && (
   260→              <div className="flex gap-1.5 px-6 pt-5 sm:px-8" aria-hidden>
   261→                {[0, 1, 2].map((s) => (
   262→                  <div
   263→                    key={s}
   264→                    className={cn(
   265→                      "h-1.5 flex-1 rounded-full transition-colors duration-300",
   266→                      phase === "success" || s <= etapeIdx ? "bg-primary" : "bg-stone-200"
   267→                    )}
   268→                  />
   269→                ))}
   270→              </div>
   271→            )}
   272→
   273→            {/* Bouton discret Quitter (étapes 1 à 3) */}
   274→            {onExit && (phase === "step1" || phase === "step2" || phase === "step3") && (
   275→              <Button
   276→                variant="ghost"
   277→                size="icon"
   278→                onClick={onExit}
   279→                aria-label="Quitter et revenir en arrière"
   280→                className="absolute right-3 top-4 z-10 size-11 rounded-full text-stone-400 hover:text-stone-600"
   281→              >
   282→                <X className="size-5" aria-hidden />
   283→              </Button>
   284→            )}
   285→
   286→            <div className="p-6 sm:p-8">
   287→              {/* En-tête du commerce sur toutes les étapes du formulaire */}
   288→              {pro && phase !== "error" && <EnteteCommerce pro={pro} avecMarge={phase !== "success"} />}
   289→
   290→              <AnimatePresence mode="wait" initial={false}>
   291→                <motion.div
   292→                  key={phase}
   293→                  variants={variantesEtape}
   294→                  initial="entree"
   295→                  animate="visible"
   296→                  exit="sortie"
   297→                  transition={{ duration: 0.25, ease: "easeOut" }}
   298→                >
   299→                  {/* ---------- ERREUR : token invalide ---------- */}
   300→                  {phase === "error" && (
   301→                    <div className="flex flex-col items-center gap-4 py-8 text-center">
   302→                      <div className="flex size-16 items-center justify-center rounded-full bg-rose-50">
   303→                        <AlertCircle className="size-9 text-rose-500" aria-hidden />
   304→                      </div>
   305→                      <div>
   306→                        <h1 className="text-xl font-bold text-stone-900">QR code invalide ou expiré</h1>
   307→                        <p className="mt-1 text-sm text-muted-foreground">
   308→                          Ce lien de scan n&apos;est plus actif. Vérifiez le QR code affiché chez le
   309→                          commerçant.
   310→                        </p>
   311→                      </div>
   312→                      <Button
   313→                        onClick={chargerScan}
   314→                        className="h-12 min-w-40 text-base font-semibold"
   315→                        aria-label="Réessayer le chargement"
   316→                      >
   317→                        Réessayer
   318→                      </Button>
   319→                      {onExit && (
   320→                        <Button
   321→                          variant="ghost"
   322→                          onClick={onExit}
   323→                          className="min-h-11 text-sm text-muted-foreground"
   324→                        >
   325→                          Retour à l&apos;accueil
   326→                        </Button>
   327→                      )}
   328→                    </div>
   329→                  )}
   330→
   331→                  {/* ---------- ÉTAPE 1 : la note ---------- */}
   332→                  {phase === "step1" && pro && (
   333→                    <div className="flex flex-col items-center text-center">
   334→                      <h1 className="text-balance text-2xl font-bold leading-snug text-stone-900">
   335→                        Comment s&apos;est passée votre visite chez{" "}
   336→                        <span className="text-primary">{pro.nomCommerce}</span> ?
   337→                      </h1>
   338→
   339→                      <div
   340→                        role="radiogroup"
   341→                        aria-label="Votre note de 1 à 5 étoiles"
   342→                        className="my-8 flex justify-center gap-1 sm:gap-2"
   343→                      >
   344→                        {[1, 2, 3, 4, 5].map((i) => (
   345→                          <motion.button
   346→                            key={i}
   347→                            type="button"
   348→                            role="radio"
   349→                            aria-checked={note === i}
   350→                            aria-label={`Noter ${i} sur 5`}
   351→                            onClick={() => choisirNote(i)}
   352→                            onKeyDown={(e) => etoileClavier(e, note)}
   353→                            whileHover={{ scale: 1.12 }}
   354→                            whileTap={{ scale: 0.88 }}
   355→                            animate={pulse === i ? { scale: [1, 1.3, 1] } : { scale: 1 }}
   356→                            transition={{ duration: 0.35, ease: "easeInOut" }}
   357→                            className="flex size-14 items-center justify-center rounded-full outline-none focus-visible:ring-2 focus-visible:ring-ring focus-visible:ring-offset-2"
   358→                          >
   359→                            <Star
   360→                              aria-hidden
   361→                              strokeWidth={1.5}
   362→                              className={cn(
   363→                                "size-12 transition-colors duration-150",
   364→                                i <= note ? "fill-amber-400 text-amber-400" : "fill-stone-200 text-stone-300"
   365→                              )}
   366→                            />
   367→                          </motion.button>
   368→                        ))}
   369→                      </div>
   370→
   371→                      <p className="text-sm text-muted-foreground">30 secondes suffisent, merci !</p>
   372→                    </div>
   373→                  )}
   374→
   375→                  {/* ---------- ÉTAPE 2 : les tags dynamiques ---------- */}
   376→                  {phase === "step2" && pro && (
   377→                    <div>
   378→                      <h1 className="text-2xl font-bold text-stone-900">
   379→                        {estPositif ? "Qu'est-ce qui vous a plu ?" : "Qu'est-ce qui s'est passé ?"}
   380→                      </h1>
   381→                      <p className="mt-1 text-sm text-muted-foreground">
   382→                        Sélectionnez tout ce qui s&apos;applique (optionnel)
   383→                      </p>
   384→
   385→                      <div
   386→                        role="group"
   387→                        aria-label={estPositif ? "Points positifs" : "Points à améliorer"}
   388→                        className="mt-5 grid grid-cols-2 gap-3"
   389→                      >
   390→                        {tagsDisponibles.map((t) => {
   391→                          const actif = tags.includes(t.id)
   392→                          return (
   393→                            <button
   394→                              key={t.id}
   395→                              type="button"
   396→                              aria-pressed={actif}
   397→                              onClick={() => basculerTag(t.id)}
   398→                              className={cn(
   399→                                "flex min-h-12 min-w-0 items-center gap-2 rounded-2xl border-2 px-3 py-3 text-left text-sm font-medium transition-all active:scale-95 sm:text-base",
   400→                                "outline-none focus-visible:ring-2 focus-visible:ring-ring focus-visible:ring-offset-2",
   401→                                actif
   402→                                  ? estPositif
   403→                                    ? "border-primary bg-primary/10 text-primary"
   404→                                    : "border-rose-500 bg-rose-50 text-rose-700"
   405→                                  : "border-stone-200 bg-white text-stone-700 hover:border-stone-300 hover:bg-stone-50"
   406→                              )}
   407→                            >
   408→                              <span aria-hidden className="text-lg leading-none">
   409→                                {t.emoji}
   410→                              </span>
   411→                              <span className="min-w-0 whitespace-nowrap leading-tight">{t.label}</span>
   412→                            </button>
   413→                          )
   414→                        })}
   415→                      </div>
   416→
   417→                      {/* Champ optionnel : personnel concerné */}
   418→                      <div className="mt-5">
   419→                        <label
   420→                          htmlFor="personnel"
   421→                          className="mb-1.5 block text-sm font-medium text-stone-700"
   422→                        >
   423→                          Personnel concerné{" "}
   424→                          <span className="font-normal text-muted-foreground">(optionnel)</span>
   425→                        </label>
   426→                        {pro.employes.length > 0 ? (
   427→                          <Select value={employeId} onValueChange={setEmployeId}>
   428→                            <SelectTrigger
   429→                              id="personnel"
   430→                              className="h-12 w-full text-base"
   431→                              aria-label="Sélectionner le personnel concerné"
   432→                            >
   433→                              <SelectValue placeholder="Personnel concerné (optionnel)" />
   434→                            </SelectTrigger>
   435→                            <SelectContent>
   436→                              {pro.employes.map((e) => (
   437→                                <SelectItem key={e.id} value={e.id}>
   438→                                  {e.nom}
   439→                                  {e.poste ? ` — ${e.poste}` : ""}
   440→                                </SelectItem>
   441→                              ))}
   442→                              <SelectItem value="autre">Autre / je ne sais pas</SelectItem>
   443→                            </SelectContent>
   444→                          </Select>
   445→                        ) : (
   446→                          <Input
   447→                            id="personnel"
   448→                            value={personnelLibre}
   449→                            onChange={(e) => setPersonnelLibre(e.target.value)}
   450→                            placeholder="Ex : la serveuse du midi"
   451→                            maxLength={120}
   452→                            className="h-12 text-base"
   453→                          />
   454→                        )}
   455→                      </div>
   456→
   457→                      <Button
   458→                        onClick={() => setPhase("step3")}
   459→                        className="mt-6 h-14 w-full text-lg font-semibold"
   460→                      >
   461→                        Continuer
   462→                      </Button>
   463→                      <Button
   464→                        variant="ghost"
   465→                        onClick={() => {
   466→                          setPulse(null)
   467→                          setPhase("step1")
   468→                        }}
   469→                        className="mt-1 min-h-11 w-full text-sm text-muted-foreground"
   470→                      >
   471→                        <ArrowLeft aria-hidden className="size-4" /> Modifier la note
   472→                      </Button>
   473→                    </div>
   474→                  )}
   475→
   476→                  {/* ---------- ÉTAPE 3 : commentaire + envoi ---------- */}
   477→                  {phase === "step3" && pro && (
   478→                    <div>
   479→                      <h1 className="text-2xl font-bold text-stone-900">Dernière étape 💬</h1>
   480→                      <p className="mt-1 text-sm text-muted-foreground">
   481→                        Un mot à nous dire ? (optionnel)
   482→                      </p>
   483→
   484→                      {/* Rappel de la sélection de tags */}
   485→                      {tags.length > 0 && (
   486→                        <p className="mt-3 text-sm text-muted-foreground">
   487→                          Votre sélection : {tags.map(tagLabel).join(" · ")}
   488→                        </p>
   489→                      )}
   490→
   491→                      <Textarea
   492→                        value={commentaire}
   493→                        onChange={(e) => setCommentaire(e.target.value)}
   494→                        rows={5}
   495→                        maxLength={2000}
   496→                        aria-label="Votre commentaire (optionnel)"
   497→                        placeholder={
   498→                          estPositif
   499→                            ? "Ce qui vous a marqué..."
   500→                            : "Décrivez ce qui s'est passé, la direction sera alertée immédiatement"
   501→                        }
   502→                        className="mt-4 min-h-28 text-base"
   503→                      />
   504→
   505→                      {/* Erreur d'envoi : message inline + réessayer */}
   506→                      {erreurEnvoi && (
   507→                        <div
   508→                          role="alert"
   509→                          className="mt-3 flex items-start gap-2 rounded-xl border border-rose-200 bg-rose-50 p-3 text-sm text-rose-700"
   510→                        >
   511→                          <AlertCircle className="mt-0.5 size-4 shrink-0" aria-hidden />
   512→                          <div>
   513→                            <p>{erreurEnvoi}</p>
   514→                            <Button
   515→                              variant="ghost"
   516→                              size="sm"
   517→                              onClick={() => envoyer(dernierCanal)}
   518→                              className="mt-1 h-9 px-2 text-sm font-semibold text-rose-700 underline underline-offset-2 hover:bg-rose-100 hover:text-rose-800"
   519→                            >
   520→                              Réessayer
   521→                            </Button>
   522→                          </div>
   523→                        </div>
   524→                      )}
   525→
   526→                      <div className="mt-4 flex flex-col gap-2.5">
   527→                        {/* Canal WhatsApp : seulement si le numéro est configuré */}
   528→                        {pro.whatsapp && (
   529→                          <Button
